        file_path = Path(file_path_str)

        else:
            print(f"⚠️ Ya existe (omitido): {file_path}")
            
    print("\n✅ ¡Estructura de directorios y archivos generada exitosamente!")

import os

OUTPUT_FILE = "proyecto_completo.txt"
VALID_EXTENSIONS = (".py", ".env", ".ini", ".md", ".json", ".txt")
EXCLUDE_DIRS = {".venv", "__pycache__", ".git", ".pytest_cache", "alembic/versions"}

def export_project():
    print("Iniciando empaquetado del proyecto...")
    count = 0
    with open(OUTPUT_FILE, "w", encoding="utf-8", errors="ignore") as outfile:
        for root, dirs, files in os.walk("."):
            # Excluimos carpetas no deseadas modificando dirs in-place
            dirs[:] = [d for d in dirs if d not in EXCLUDE_DIRS]
            
            for file in files:
                if file.endswith(VALID_EXTENSIONS) and file != OUTPUT_FILE:
                    file_path = os.path.join(root, file)
                    print(f"Procesando: {file_path}")
                    
                    outfile.write(f"\n\n{'='*80}\n")
                    outfile.write(f"ARCHIVO: {file_path}\n")
                    outfile.write(f"{'='*80}\n\n")
                    
                    try:
                        with open(file_path, "r", encoding="utf-8", errors="ignore") as infile:
                            outfile.write(infile.read())
                        count += 1
                    except Exception as e:
                        outfile.write(f"[Error al leer el archivo: {e}]\n")
                        
    print(f"\n¡Éxito total! Se empaquetaron {count} archivos en: {OUTPUT_FILE}")

if __name__ == "__main__":
    export_project()

# ALEMBIC ENVIRONMENT CONFIGURATION (alembic/env.py)
# Description: Configures Alembic to read database credentials dynamically from 
# our Pydantic settings and registers SQLAlchemy models for 'autogenerate' support.

from logging.config import fileConfig
from sqlalchemy import engine_from_config
from sqlalchemy import pool
from alembic import context

# 1. Import application settings and declarative Base containing our ORM models
from app.core.config import settings
from app.core.database import Base

# Import all models explicitly so Alembic detects them during autogeneration
from app.models.user import User
from app.models.item import Item

# This is the Alembic Config object, which provides access to values within alembic.ini
config = context.config

# Interpret the config file for Python logging if a config file is present
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# 2. Overwrite SQLAlchemy URL in Alembic config with our dynamic Pydantic settings URL
config.set_main_option("sqlalchemy.url", settings.DATABASE_URL)

# 3. Set target_metadata for 'autogenerate' support (enables Alembic to detect model changes)
target_metadata = Base.metadata

def run_migrations_offline() -> None:
    """
    Run migrations in 'offline' mode.
    Outputs SQL statements directly to stdout without executing them on a live engine.
    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()

def run_migrations_online() -> None:
    """
    Run migrations in 'online' mode.
    Creates an Engine and associates a connection with the transaction context.
    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

# ==============================================================================
# FILE: app/api/deps.py
# DESCRIPTION: Shared API dependencies (Dependency Injection pattern).
# Encapsulates reusable logic such as database session yielding and current 
# authenticated user resolution via JWT tokens.
# ==============================================================================

from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
from sqlalchemy.orm import Session

from app.core.config import settings
from app.core.database import get_db  # <--- Importamos get_db directamente desde database.py
from app.models.user import User as UserModel

# OAuth2 scheme definition pointing to the token generation endpoint.
# This tells FastAPI where clients should send credentials to obtain a token.
reusable_oauth2 = OAuth2PasswordBearer(
    tokenUrl=f"{settings.API_V1_STR}/auth/login"
)


def get_current_user(
    db: Session = Depends(get_db), token: str = Depends(reusable_oauth2)
) -> UserModel:
    """
    Validates the incoming JWT bearer token, extracts the subject (user email/id),
    and queries the database to return the active user instance.
    Implements a fail-secure pattern raising 401 Unauthorized if credentials fail.
    """
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        # Decode the JWT token using our secret key and algorithm
        payload = jwt.decode(
            token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM]
        )
        user_email: str = payload.get("sub")
        if user_email is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception

    # Query the user in the database
    user = db.query(UserModel).filter(UserModel.email == user_email).first()
    if user is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, detail="User not found"
        )
    
    return user

# ==============================================================================
# FILE: app/api/v1/endpoints/auth.py
# DESCRIPTION: Authentication and Login endpoints (OAuth2 implementation).
# Handles identity verification and JWT access token generation.
# ==============================================================================

from datetime import timedelta
from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.orm import Session

from app.core.config import settings
from app.core.database import get_db
from app.core.security import verify_password, create_access_token
from app.models.user import User as UserModel
from app.schemas.token import Token

router = APIRouter()


@router.post("/login/access-token", response_model=Token)
def login_access_token(
    db: Session = Depends(get_db),
    form_data: OAuth2PasswordRequestForm = Depends()
):
    """
    OAuth2 compatible token login endpoint. Verifies user credentials and returns a JWT bearer token.
    - **form_data.username**: Expected to be the user's email address.
    - **form_data.password**: Plain text password to verify against hashed database record.
    """
    # 1. Query user by email (username field in OAuth2 spec)
    user = db.query(UserModel).filter(UserModel.email == form_data.username).first()
    
    # 2. Validate user existence and password correctness
    if not user or not verify_password(form_data.password, user.hashed_password):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Incorrect email or password",
        )
    
    # 3. Check if user account is active
    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Inactive user account",
        )
    
    # 4. Generate expiration delta and sign JWT access token
    access_token_expires = timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    access_token = create_access_token(
        subject=user.email, expires_delta=access_token_expires
    )
    
    return {
        "access_token": access_token,
        "token_type": "bearer",
    }

# ==============================================================================
# FILE: app/api/v1/endpoints/items.py
# DESCRIPTION: Item management endpoints (Protected CRUD with JWT).
# Handles HTTP routes to create and list items linked exclusively to the 
# authenticated user context.
# ==============================================================================

from typing import List
from fastapi import APIRouter, Depends, status
from sqlalchemy.orm import Session

from app.api.deps import get_current_user
from app.core.database import get_db
from app.models.item import Item as ItemModel
from app.models.user import User as UserModel
from app.schemas.item import Item, ItemCreate

router = APIRouter()


@router.post("/", response_model=Item, status_code=status.HTTP_201_CREATED)
def create_item(
    item_in: ItemCreate,
    db: Session = Depends(get_db),
    current_user: UserModel = Depends(get_current_user),
):
    """
    Create a new item associated automatically with the currently authenticated user.
    - **item_in**: Pydantic payload containing title and description.
    - **db**: Active database session injection.
    - **current_user**: Validated user model extracted from JWT token.
    """
    # Unpack validated Pydantic model data and inject the owner's foreign key reference
    db_item = ItemModel(**item_in.model_dump(), owner_id=current_user.id)
    
    db.add(db_item)
    db.commit()
    db.refresh(db_item)  # Refresh to retrieve database-generated fields (id, timestamps)
    
    return db_item


@router.get("/", response_model=List[Item])
def read_items(
    skip: int = 0,
    limit: int = 100,
    db: Session = Depends(get_db),
    current_user: UserModel = Depends(get_current_user),
):
    """
    Retrieve a paginated list of items belonging exclusively to the authenticated user.
    - **skip**: Offset for pagination (default: 0).
    - **limit**: Maximum number of records to return (default: 100).
    """
    items = (
        db.query(ItemModel)
        .filter(ItemModel.owner_id == current_user.id)
        .offset(skip)
        .limit(limit)
        .all()
    )
    return items

# Description: Centralizes application settings and environment variables validation
# using Pydantic v2 Settings. Dynamically assembles the DATABASE_URL to avoid 
# hardcoded ports and enforces 'Fail Fast' validation at startup.
# ==============================================================================

from pydantic import computed_field
from pydantic_settings import BaseSettings, SettingsConfigDict

class Settings(BaseSettings):
    """
    Settings class utilizing Pydantic BaseSettings to parse environment variables
    from the .env file with strict runtime type checking and dynamic assembly.
    """
    # Application metadata
    PROJECT_NAME: str = "Premium Backend API (v2)"
    API_V1_STR: str = "/api/v1"  # <--- Añadimos esta línea faltante
    
    # Database individual parameters (Decoupled configuration)
    POSTGRES_SERVER: str = "localhost"
    POSTGRES_PORT: int = 5432
    POSTGRES_USER: str
    POSTGRES_PASSWORD: str
    POSTGRES_DB: str

    # Computed field to dynamically assemble SQLAlchemy connection string safely
    @computed_field
    @property
    def DATABASE_URL(self) -> str:
        """
        Dynamically constructs the PostgreSQL connection URL using decoupled parameters,
        preventing hardcoded values and supporting parallel local environments.
        """
        return f"postgresql+psycopg2://{self.POSTGRES_USER}:{self.POSTGRES_PASSWORD}@{self.POSTGRES_SERVER}:{self.POSTGRES_PORT}/{self.POSTGRES_DB}"
    
    # Security & JWT configuration
    SECRET_KEY: str
    ALGORITHM: str = "HS256"
    ACCESS_TOKEN_EXPIRE_MINUTES: int = 30

    # Pydantic v2 configuration to point to our local .env file
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore" # Ignores extra variables present in .env not defined here
    )

# Instantiate settings to be imported across the application
settings = Settings()

"""
================================================================================
MODULE: Security, Hashing, and JWT Authentication (app/core/security.py)
================================================================================
Description:
    This module centralizes all cryptographic logic for the application:
    1. Unidirectional password hashing using the 'bcrypt' library directly 
       (mitigating compatibility issues found in legacy wrappers).
    2. Generation, signing, and decoding of JSON Web Tokens (JWT) to enforce 
       a stateless authentication architecture.
================================================================================
"""

from datetime import datetime, timedelta
from typing import Optional, Union, Any
from jose import jwt
import bcrypt

from app.core.config import settings

# ==============================================================================
# PASSWORD HASHING FUNCTIONS
# ==============================================================================

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """
    Verify a plain text password against a securely hashed counterpart.
    
    Args:
        plain_password (str): The raw password provided by the user during login.
        hashed_password (str): The secure hash stored in the database.
        
    Returns:
        bool: True if passwords match, False otherwise.
    """
    # bcrypt requires bytes; encode strings accordingly
    password_byte_enc = plain_password.encode('utf-8')
    hashed_password_byte_enc = hashed_password.encode('utf-8')
    
    return bcrypt.checkpw(password_byte_enc, hashed_password_byte_enc)


def get_password_hash(password: str) -> str:
    """
    Generate a secure cryptographic hash from a plain text password using bcrypt.
    
    Security Note: 
    Bcrypt has a native hard limit of 72 bytes maximum. To prevent system errors 
    or unpredictable behavior with excessively long passwords, we encode the string 
    to bytes and apply a preventive truncation at the first 72 bytes.
    
    Args:
        password (str): The raw password to be encrypted.
        
    Returns:
        str: The resulting secure hash string ready to be stored in the database.
    """
    pwd_bytes = password.encode('utf-8')
    if len(pwd_bytes) > 72:
        pwd_bytes = pwd_bytes[:72]
        
    # Generate a random salt and hash the password using bcrypt directly
    hashed_password = bcrypt.hashpw(pwd_bytes, bcrypt.gensalt())
    
    return hashed_password.decode('utf-8')


# ==============================================================================
# JSON WEB TOKEN (JWT) FUNCTIONS
# ==============================================================================

def create_access_token(subject: Union[str, Any], expires_delta: Optional[timedelta] = None) -> str:
    """
    Create a signed JSON Web Token (JWT) for stateless user authorization.
    
    Args:
        subject (Union[str, Any]): The unique identifier for the token payload (typically user email/id).
        expires_delta (Optional[timedelta]): Custom expiration time window. Defaults to system settings.
        
    Returns:
        str: Encoded JWT access token string.
    """
    # Handle token expiration calculation with fallback logic:
    if expires_delta:
        # Custom expiration scenario: Executed when an explicit argument is passed.
        # Use Case: Useful for features like "Remember Me" (e.g., 7 days expiration) 
        # or short-lived tokens like password reset tokens (e.g., 15 minutes).
        expire = datetime.utcnow() + expires_delta
    else:
        # Default expiration scenario: Executed when no custom delta is specified.
        # Fallbacks automatically to the standard configuration defined in settings 
        # (settings.ACCESS_TOKEN_EXPIRE_MINUTES, typically 30 minutes for standard logins).
        expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
        
    # Construct the payload dictionary
    to_encode = {"exp": expire, "sub": str(subject)}
    
    # Encode and sign the JWT using the secret key and algorithm defined in settings
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    
    return encoded_jwt

¡Excelente pregunta técnica! Como tu mentor, me encanta que anticipes estos escenarios de infraestructura local.

Sí, **habría un conflicto directo** si intentas ejecutar exactamente la misma línea de comandos tal cual está escrita, por dos motivos principales a nivel de Docker:

1. **Conflicto por el nombre del contenedor (`--name postgres-db`):** Docker no permite que existan dos contenedores activos o detenidos con exactamente el mismo nombre en todo tu sistema. Si el proyecto anterior ya usó el nombre `postgres-db`, Docker arrojará un error diciendo que el nombre ya está en uso.
2. **Conflicto por el puerto (`-p 5432:5432`):** El puerto `5432` de tu máquina física anfitriona ya estaría asociado o reservado por el contenedor anterior si este se encuentra encendido.

---

### ¿Cómo lo resolvemos profesionalmente?

Dado que estás corriendo dos versiones de un proyecto y quieres que convivan sin pisarse, tienes dos opciones excelentes:

#### Opción A: Aislar el nuevo contenedor con otro nombre y puerto (Recomendado para desarrollo paralelo)

Puedes cambiar el nombre del contenedor y el puerto externo para que el proyecto 2 corra en paralelo sin interferir con el proyecto 1:

```bash
docker run --name postgres-db-v2 -e POSTGRES_USER=postgres -e POSTGRES_PASSWORD=postgres -e POSTGRES_DB=backend_db_v2 -p 5433:5432 -d postgres:15

```

* **¿Qué cambió aquí?**
* Le pusimos `--name postgres-db-v2` para distinguirlo del anterior.
* Cambiamos la base de datos interna a `backend_db_v2`.
* Mapeamos el puerto externo a `5433:5432` (tu PC escuchará en el puerto 5433 y lo redirigirá al 5432 interno del contenedor).
* *Nota si usas esta opción:* En tu archivo `.env` del nuevo proyecto, deberás actualizar la URL de conexión apuntando al puerto nuevo: `localhost:5433/backend_db_v2`.



#### Opción B: Apagar y reciclar el contenedor anterior (Si ya no usarás el proyecto viejo)

Si el proyecto anterior ya cumplió su ciclo y prefieres usar exactamente la misma configuración y el mismo puerto (`5432`), simplemente asegúrate de detener y eliminar el contenedor viejo antes de levantar el nuevo:

```bash
# Detener el contenedor viejo
docker stop postgres-db

# Eliminar el contenedor viejo (libera el nombre y los puertos)
docker rm postgres-db

```

Una vez hecho esto, ya podrías ejecutar tu comando original de la versión 1 sin ningún conflicto.

En arquitecturas limpias, es una excelente práctica utilizar variables en los archivos `.env` (como `DB_PORT=5432`) en lugar de hardcodear los puertos, facilitando cambiar de entorno con solo modificar una línea de configuración.
